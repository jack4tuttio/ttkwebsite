import { readFile, writeFile } from 'node:fs/promises';

const INPUT = new URL('../flagged.txt', import.meta.url);
const OUTPUT = new URL('../data.json', import.meta.url);
const FALLBACK_AVATAR = 'data:image/svg+xml;charset=UTF-8,' + encodeURIComponent(`
<svg xmlns="http://www.w3.org/2000/svg" width="180" height="180"><rect width="100%" height="100%" fill="#232936"/><text x="50%" y="54%" text-anchor="middle" fill="#9ba4b3" font-size="54" font-family="Arial">?</text></svg>`);

function parseFlaggedFile(text) {
  return text.split(/\r?\n/)
    .map(line => line.trim())
    .filter(line => line && !line.startsWith('#'))
    .map((line, index) => {
      const parts = line.split('|').map(v => v.trim());
      const [userId, categories, summary, evidence = '', added = 'Unknown', status = 'Flagged'] = parts;
      if (!/^\d+$/.test(userId || '') || !categories || !summary) {
        throw new Error(`Invalid flagged.txt entry on data line ${index + 1}. The first field must be a numeric Roblox user ID.`);
      }
      return {
        userId,
        categories: categories.split(',').map(v => v.trim()).filter(Boolean),
        summary,
        evidence: evidence.split(',').map(v => v.trim()).filter(Boolean),
        added,
        status
      };
    });
}

async function fetchJson(url) {
  const response = await fetch(url, {
    headers: { 'User-Agent': 'RobloxFlagged-GitHub-Pages-Builder/1.0' }
  });
  if (!response.ok) throw new Error(`${response.status} ${response.statusText}`);
  return response.json();
}

async function getProfile(userId) {
  try {
    return await fetchJson(`https://users.roblox.com/v1/users/${encodeURIComponent(userId)}`);
  } catch (error) {
    console.warn(`Profile lookup failed for ${userId}: ${error.message}`);
    return {
      id: Number(userId),
      name: `User ${userId}`,
      displayName: `User ${userId}`,
      created: null,
      description: ''
    };
  }
}

async function getAvatars(userIds) {
  const result = new Map();
  const chunkSize = 100;
  for (let i = 0; i < userIds.length; i += chunkSize) {
    const chunk = userIds.slice(i, i + chunkSize);
    try {
      const url = `https://thumbnails.roblox.com/v1/users/avatar-headshot?userIds=${chunk.join(',')}&size=180x180&format=Png&isCircular=false`;
      const payload = await fetchJson(url);
      for (const item of payload.data || []) {
        if (item.imageUrl) result.set(String(item.targetId), item.imageUrl);
      }
    } catch (error) {
      console.warn(`Thumbnail lookup failed for batch: ${error.message}`);
    }
  }
  return result;
}

const reports = parseFlaggedFile(await readFile(INPUT, 'utf8'));
const uniqueIds = [...new Set(reports.map(r => r.userId))];
const profiles = new Map(await Promise.all(uniqueIds.map(async id => [id, await getProfile(id)])));
const avatars = await getAvatars(uniqueIds);

const output = reports.map(report => ({
  ...report,
  profile: profiles.get(report.userId),
  avatar: avatars.get(report.userId) || FALLBACK_AVATAR
}));

await writeFile(OUTPUT, JSON.stringify({ generatedAt: new Date().toISOString(), reports: output }, null, 2) + '\n');
console.log(`Generated data.json with ${output.length} report(s).`);
